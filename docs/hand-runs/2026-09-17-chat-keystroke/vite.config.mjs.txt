import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import { fileURLToPath } from "node:url";

// A production build of the bench harness only: same React, same
// Markdown.jsx, same TaskChat.jsx as the app, without the app's own
// shell around it. Kept out of ui/vite.config.js so `npm run build`
// still writes exactly what pkg/ui embeds.
export default defineConfig({
  root: fileURLToPath(new URL(".", import.meta.url)),
  plugins: [react()],
  build: {
    outDir: fileURLToPath(new URL("./dist", import.meta.url)),
    emptyOutDir: true,
  },
});
