// The driver behind docs/hand-runs/2026-09-17-chat-keystroke.md: what one
// keystroke in a task's chat costs, against a conversation of N messages.
//
// Chromium's own Performance.getMetrics over CDP, the same measurement
// docs/hand-runs/2026-09-07-panel-dashboard.md took of the poll -- script
// time is the number that matters here, because nothing is fetched
// between one character and the next and every millisecond of the delay
// is on the main thread.
//
// Run it from ui/, where @playwright/test resolves:
//
//   npx vite build --config bench/vite.config.mjs
//   npx vite preview --config bench/vite.config.mjs --port 5199 &
//   node bench/drive.mjs http://127.0.0.1:5199
import { chromium } from "@playwright/test";

const base = process.argv[2] || "http://127.0.0.1:5199";
const COUNTS = (process.env.BENCH_N || "5,20,40,80")
  .split(",")
  .map((s) => Number(s.trim()));
const KEYS = Number(process.env.BENCH_KEYS || 20);
const POLLS = Number(process.env.BENCH_POLLS || 10);
const COMPOSER = ".cs-message-input__content-editor";

async function metrics(client) {
  const { metrics } = await client.send("Performance.getMetrics");
  const m = Object.fromEntries(metrics.map((x) => [x.name, x.value]));
  return {
    script: m.ScriptDuration,
    layout: m.LayoutDuration,
    ts: m.Timestamp,
  };
}

const delta = (a, b) => ({
  script: (b.script - a.script) * 1000,
  layout: (b.layout - a.layout) * 1000,
  wall: (b.ts - a.ts) * 1000,
});

const browser = await chromium.launch();
const rows = [];

for (const n of COUNTS) {
  const ctx = await browser.newContext({
    viewport: { width: 1440, height: 900 },
  });
  const page = await ctx.newPage();
  const client = await ctx.newCDPSession(page);
  await client.send("Performance.enable");
  await page.goto(`${base}/?n=${n}`, { waitUntil: "load" });
  await page.waitForSelector(COMPOSER);
  await page.waitForFunction("window.__bench && window.__bench.poll");
  const bodies = await page.locator(".task-chat-body").count();

  // Warm up: the first keystroke pays for whatever is still lazy.
  await page.click(COMPOSER);
  await page.keyboard.type("warm", { delay: 20 });

  const beforeKeys = await metrics(client);
  for (let i = 0; i < KEYS; i++) {
    await page.keyboard.type("a");
    // One frame per character, so the commit and the paint it forces are
    // inside the window being measured rather than after it.
    await page.evaluate(() => new Promise(requestAnimationFrame));
  }
  const keys = delta(beforeKeys, await metrics(client));

  const beforePolls = await metrics(client);
  for (let i = 0; i < POLLS; i++) {
    await page.evaluate((i) => window.__bench.poll(i), i);
    await page.evaluate(() => new Promise(requestAnimationFrame));
  }
  const polls = delta(beforePolls, await metrics(client));

  rows.push({
    messages: n + 1,
    bodies,
    keyScript: keys.script / KEYS,
    keyWall: keys.wall / KEYS,
    pollScript: polls.script / POLLS,
    pollWall: polls.wall / POLLS,
  });
  await ctx.close();
}

await browser.close();

console.log(
  [
    "messages",
    "bodies",
    "script/key",
    "wall/key",
    "script/poll",
    "wall/poll",
  ].join("\t"),
);
for (const r of rows) {
  console.log(
    [
      r.messages,
      r.bodies,
      r.keyScript.toFixed(1),
      r.keyWall.toFixed(1),
      r.pollScript.toFixed(1),
      r.pollWall.toFixed(1),
    ].join("\t"),
  );
}
